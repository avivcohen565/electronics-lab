\section{שיעור 1: דיודות}
\subsection*{משוואת הדיודה}
\textbf{משוואת שוקלי}
\[I_D = I_S\left(e^{\frac{V_D}{nV_T}}-1\right)\]
\textbf{ממתח קדמי}
\[I_D \approx I_S\, e^{\frac{V_D}{nV_T}}\]
\textbf{ממתח אחורי}
\[I_D \approx -I_S\]
\textbf{מתח תרמי}
\[V_T = \frac{kT}{q} \approx 25\,\text{mV}\]
\textbf{תוספת מתח לכל פי 10 בזרם}
\[\Delta V_D = nV_T\ln 10 \approx 58n\,\text{mV}\]
\subsection*{מודלים}
\textbf{דיודה אידיאלית}
\[\begin{cases} V_D>0 & \Rightarrow \text{short} \\ V_D<0 & \Rightarrow \text{open},\; I_D=0 \end{cases}\]
\textbf{מודל שני (מקור מתח)}
\[i_D=\begin{cases}0 & v_D<V_{D0}\\ \infty & v_D>V_{D0}\end{cases},\qquad V_{D0}\approx 0.5\text{–}0.7\,\text{V}\]
\textbf{מודל שלישי (מקור מתח והתנגדות)}
\[i_D=\begin{cases}0 & v_D\le V_{D0}\\ \dfrac{v_D-V_{D0}}{r_D} & v_D>V_{D0}\end{cases}\]
\subsection*{קו עבודה ואות קטן}
\textbf{קו עבודה}
\[I_D = \frac{V_{DC}-V_D}{R} = -\frac{1}{R}V_D+\frac{V_{DC}}{R}\]
\textbf{נקודת עבודה}
\[Q = \left(V_{DQ},\, I_{DQ}\right)\]
\textbf{פירוק DC ו-AC}
\[v_D(t)=V_D+v_d(t),\qquad i_D(t)=I_D+i_d(t)\]
\textbf{פיתוח אות קטן}
\[i_D = I_D\,e^{\frac{v_d}{nV_T}} \approx I_D\left(1+\frac{v_d}{nV_T}\right)=I_D+\frac{v_d}{r_d}\]
\textbf{התנגדות דינמית}
\[r_d = \left.\frac{dv_D}{di_D}\right|_Q = \frac{nV_T}{I_{DQ}}\]
\textbf{זרם AC}
\[i_d(t)=\frac{v_d(t)}{r_d}\]
\subsection*{דיודת זנר}
\textbf{מודל זנר}
\[V_Z = V_{Z0}+r_Z I_Z,\qquad r_Z=\frac{\Delta V}{\Delta I}\]
\textbf{הספק}
\[P_Z = I_Z V_Z \approx I_Z V_{Z0},\qquad P_{Z,\max}=I_{Z,\max}V_{Z,\max}\]
\textbf{KCL במייצב}
\[I = I_Z + I_L,\qquad I=\frac{V_S-V_Z}{R},\qquad I_L=\frac{V_Z}{R_L}\]
\textbf{נגד טורי מקסימלי}
\[R_{\max}=\frac{V_S-V_Z}{I_{Z,\min}+I_L}\]
\textbf{נגד טורי מינימלי}
\[R_{\min}=\frac{V_S-V_Z}{I_{Z,\max}+I_L}\]
\textbf{מתח מוצא}
\[V_{out}=V_{Z0}\frac{R\parallel R_L}{r_Z+R\parallel R_L}+V_S\frac{r_Z\parallel R_L}{R+r_Z\parallel R_L}\]
\textbf{גלית במוצא}
\[\Delta V_{out}=\Delta V_S\,\frac{r_Z\parallel R_L}{r_Z\parallel R_L+R}\]
\section{שיעור 2: מיישרי גל}
\subsection*{שנאי}
\textbf{מקדם השנאה}
\[a=\frac{N_P}{N_S}=\frac{V_P}{V_S}=\frac{I_S}{I_P}\]
\textbf{ערך שיא}
\[V_{peak}=\sqrt{2}\,V_{rms}\]
\subsection*{מיישר חצי גל}
\textbf{מוצא (מודל מעשי)}
\[v_{out}=\begin{cases}0 & v_S<V_{D0}\\ \dfrac{R}{R+r_D}\left(v_S-V_{D0}\right) & v_S\ge V_{D0}\end{cases}\]
\textbf{ממוצע ויעיל}
\[V_{out,DC}=\frac{V_{O,P}}{\pi},\qquad V_{out,RMS}=\frac{V_{O,P}}{2}\]
\textbf{PIV}
\[\text{PIV}=V_S\]
\subsection*{גל שלם עם סנף מרכזי}
\textbf{שיא המוצא}
\[V_{O,P}=V_S-V_D\]
\textbf{PIV}
\[\text{PIV}=2V_S-V_D\]
\textbf{ממוצע ויעיל}
\[V_{out,DC}=\frac{2V_{O,P}}{\pi},\qquad V_{out,RMS}=\frac{V_{O,P}}{\sqrt2}\]
\subsection*{גשר דיודות}
\textbf{שיא המוצא}
\[V_{O,P}=V_S-2V_D\]
\textbf{PIV}
\[\text{PIV}=V_S-V_D\]
\subsection*{סינון בקבל}
\textbf{תנאי גלאי שיא}
\[RC\gg T\]
\textbf{גלית, סנף מרכזי}
\[V_r=\frac{V_P-V_D}{2fRC}\]
\textbf{גלית, גשר}
\[V_r=\frac{V_P-2V_D}{2fRC}\]
\textbf{גלית, חצי גל}
\[V_r=\frac{V_{O,P}}{fRC}\]
\textbf{מתח DC במוצא}
\[V_{out,DC}=\left(V_P-V_D\right)-\frac{1}{2}V_r\]
\textbf{זרם עומס}
\[I_L=\frac{V_P-V_D}{R}\;\;(\text{CT}),\qquad I_L=\frac{V_P-2V_D}{R}\;\;(\text{bridge})\]
\textbf{זרם ממוצע בדיודה}
\[i_{D,AVG}=I_L\left(1+\pi\sqrt{\frac{V_P}{2V_r}}\right)\]
\textbf{זרם שיא בדיודה}
\[i_{D,\max}=I_L\left(1+2\pi\sqrt{\frac{V_P}{2V_r}}\right)\]
\section{שיעור 3: MOSFET}
\subsection*{הגדרות}
\textbf{מתח יתר}
\[V_{OV}=V_{GS}-V_T\]
\textbf{פרמטרי תהליך}
\[k_n'=\mu_n C_{ox},\qquad K_n=\frac12 k_n'\frac{W}{L}\]
\textbf{גבול הרוויה}
\[V_{DS,sat}=V_{OV}=V_{GS}-V_T\]
\subsection*{NMOS Enhancement}
\textbf{קטעון}
\[V_{GS}<V_T \;\Rightarrow\; I_D=0\]
\textbf{טריודה}
\[I_D=k_n'\frac{W}{L}\left(\left(V_{GS}-V_T\right)V_{DS}-\frac12V_{DS}^2\right),\quad V_{DS}<V_{OV}\]
\textbf{רוויה}
\[I_D=\frac12k_n'\frac{W}{L}\left(V_{GS}-V_T\right)^2=K_nV_{OV}^2,\quad V_{DS}\ge V_{OV}\]
\textbf{גבול טריודה-רוויה}
\[i_D=\frac12k_n'\frac{W}{L}\,v_{DS}^2\]
\textbf{התנגדות בטריודה ($V_{DS}$ קטן)}
\[r_{DS}\approx\frac{1}{k_n'\frac{W}{L}V_{OV}}\]
\subsection*{PMOS Enhancement}
\textbf{תנאי הולכה}
\[V_T<0,\qquad V_{GS}<V_T\]
\textbf{טריודה}
\[I_D=k_p'\frac{W}{L}\left(\left(V_{SG}-|V_T|\right)V_{SD}-\frac12V_{SD}^2\right)\]
\textbf{רוויה}
\[I_D=\frac12k_p'\frac{W}{L}\left(V_{SG}-|V_T|\right)^2\]
\subsection*{אפקטים מסדר שני}
\textbf{קיצור תעלה}
\[I_D=\frac12k_n'\frac{W}{L}\left(V_{GS}-V_T\right)^2\left(1+\lambda V_{DS}\right)\]
\textbf{מתח ארלי}
\[V_A=\frac{1}{\lambda}\]
\textbf{אפקט מצע}
\[V_T=V_{T0}+\gamma\left(\sqrt{0.6+V_{SB}}-\sqrt{0.6}\right),\qquad \gamma\approx0.5\,\sqrt{\text{V}}\]
\subsection*{מודל אות קטן}
\textbf{טרנס-מוליכות}
\[g_m=k_n'\frac{W}{L}V_{OV}=\sqrt{2k_n'\frac{W}{L}I_D}=\frac{2I_D}{V_{OV}}\]
\textbf{התנגדות מוצא}
\[r_o=\frac{1}{\lambda I_D}=\frac{V_A}{I_D}\]
\textbf{הגבר עצמי}
\[\mu=g_m r_o\]
\textbf{מודל $\pi$}
\[i_d=g_mv_{gs}+\frac{v_{ds}}{r_o},\qquad i_g=0\]
\textbf{מודל T}
\[r_s=\frac{1}{g_m}\]
\section{שיעור 4: תצורות חיבור עם MOSFET}
\subsection*{מודל מגבר כללי}
\textbf{הגבר מתח ללא עומס}
\[A_{vo}=\left.\frac{v_o}{v_i}\right|_{i_o=0}\]
\textbf{הגבר זרם בקצר}
\[A_{is}=\left.\frac{i_o}{i_i}\right|_{v_o=0}\]
\textbf{טרנס-מוליכות}
\[G_m=\left.\frac{i_o}{v_i}\right|_{v_o=0}\]
\textbf{טרנס-התנגדות}
\[R_m=\left.\frac{v_o}{i_i}\right|_{i_o=0}\]
\textbf{הגבר כולל}
\[G_v=\frac{v_o}{v_{sig}}=\frac{R_{in}}{R_{in}+R_{sig}}\,A_{vo}\,\frac{R_L}{R_L+R_o}\]
\subsection*{CS (מקור משותף)}
\textbf{התנגדות כניסה}
\[R_{in}=R_G\]
\textbf{התנגדות מוצא}
\[R_o=\left(g_mR_S+1\right)r_o+R_S,\qquad R_{out}=R_o\parallel R_D\]
\textbf{הגבר מתח}
\[A_v=-\frac{g_m\left(R_L\parallel R_D\right)}{\left(1+g_mR_S\right)+\dfrac{R_S+R_L\parallel R_D}{r_o}}\]
\textbf{$r_{o}$$\to$$\infty$}
\[A_v=-\frac{g_m\left(R_L\parallel R_D\right)}{1+g_mR_S}\]
\textbf{$R_{S}$=0}
\[A_v=-\frac{g_m\left(R_L\parallel R_D\right)}{1+\left(R_L\parallel R_D\right)/r_o}\]
\textbf{$R_{S}$=0, $r_{o}$$\to$$\infty$}
\[A_v=-g_m\left(R_L\parallel R_D\right)\]
\subsection*{CG (שער משותף)}
\textbf{התנגדות כניסה}
\[R_{in}=\frac{1+\left(R_L\parallel R_D\right)/r_o}{1/r_o+g_m}\]
\textbf{התנגדות מוצא}
\[R_o=\left(g_mr_o+1\right)\left(R_{sig}\parallel R_S\right)+r_o,\qquad R_{out}=R_o\parallel R_D\]
\textbf{הגבר מתח}
\[A_v=\frac{\left(1+g_mr_o\right)\left(R_L\parallel R_D\right)}{r_o+R_L\parallel R_D}\]
\textbf{$r_{o}$$\to$$\infty$}
\[R_{in}=\frac{1}{g_m},\quad R_o\to\infty,\quad R_{out}=R_D,\quad A_v=g_m\left(R_L\parallel R_D\right)\]
\subsection*{CD (עוקב מקור)}
\textbf{התנגדות כניסה}
\[R_{in}\to\infty\]
\textbf{התנגדות מוצא}
\[R_{out}=r_o\parallel\frac{1}{g_m}\parallel R_S=\frac{r_o\parallel R_S}{1+g_m\left(r_o\parallel R_S\right)}\]
\textbf{הגבר מתח}
\[A_v=\frac{g_m\left(r_o\parallel R_L\parallel R_S\right)}{1+g_m\left(r_o\parallel R_L\parallel R_S\right)}\approx1\]
\textbf{עם $R_{D}$}
\[A_v=\frac{g_mr_o\left(R_L\parallel R_S\right)}{\left(1+g_mr_o\right)\left(R_L\parallel R_S\right)+r_o+R_D},\qquad R_o=\frac{r_o+R_D}{1+g_mr_o}\]
\section{שיעור 5: טרנזיסטור BJT}
\subsection*{קירכהוף}
\textbf{KCL}
\[I_E=I_B+I_C\]
\textbf{KVL}
\[V_{CE}=V_{CB}+V_{BE}\;\;(\text{NPN}),\qquad V_{EC}=V_{BC}+V_{EB}\;\;(\text{PNP})\]
\subsection*{מצבי פעולה (NPN)}
\textbf{קטעון}
\[V_{BE}<0.7\,\text{V}\;\Rightarrow\;I_B=I_C=0,\;V_{CE}=V_{CC}\]
\textbf{פעיל קדמי}
\[V_{BE}=0.7\,\text{V},\;V_{CE}>V_{BE}\;\Rightarrow\;I_C=\beta I_B=\alpha I_E\]
\textbf{רוויה}
\[V_{CE}=V_{CE,sat}\approx0.2\,\text{V},\qquad I_C=I_{C,sat}<\beta I_B\]
\textbf{זרם רוויה}
\[I_{C,sat}=\frac{V_{CC}-V_{CE,sat}}{R_C}\]
\textbf{הקשר $\alpha$-$\beta$}
\[\alpha=\frac{\beta}{\beta+1},\qquad \beta=\frac{\alpha}{1-\alpha}\]
\subsection*{קו עבודה}
\textbf{KVL קולט}
\[V_{CC}=I_CR_C+V_{CE}+I_ER_E\]
\textbf{KVL בסיס}
\[V_{BB}=I_BR_B+V_{BE}+I_ER_E\]
\textbf{קו עבודה}
\[I_C=-\frac{1}{R_C}V_{CE}+\frac{V_{CC}}{R_C}\]
\subsection*{מודל אות קטן}
\textbf{טרנס-מוליכות}
\[g_m=\frac{I_C}{V_T}\]
\textbf{היבריד-$\pi$}
\[r_\pi=\frac{v_{be}}{i_b}=\frac{\beta}{g_m}=\left(\beta+1\right)r_e=\frac{\beta V_T}{I_C}\]
\textbf{מודל T}
\[r_e=\frac{\alpha}{g_m}=\frac{V_T}{I_E},\qquad i_b=\frac{v_e}{r_\pi}\]
\textbf{אפקט ארלי}
\[r_o=\frac{V_A}{I_C}\]
\subsection*{CE (פולט משותף)}
\textbf{התנגדות כניסה}
\[R_{in}=r_\pi+\frac{R_E\left(\beta+1\right)r_o}{r_o+R_E+R_L}\]
\textbf{התנגדות מוצא}
\[R_o=r_o+\left(1+g_mr_o\right)\left(r_\pi\parallel R_E\right)\]
\textbf{הגבר מתח}
\[\frac{v_o}{v_i}=-\frac{\beta R_L}{r_\pi+\left(\beta+1\right)R_E+\left(r_\pi/r_o\right)\left(R_E+R_L\right)}\]
\textbf{הגבר זרם}
\[\frac{i_o}{i_i}=-\frac{\beta}{1+\left(R_E+R_L\right)/r_o}\]
\textbf{$r_{o}$$\to$$\infty$}
\[R_{in}=r_\pi+\left(\beta+1\right)R_E,\quad \frac{v_o}{v_i}=-\frac{\beta R_L'}{r_\pi+\left(\beta+1\right)R_E},\quad \frac{i_o}{i_i}=-\beta\]
\textbf{$R_{E}$$\to$0}
\[\frac{v_o}{v_i}=-g_mR_L'\]
\subsection*{CC (עוקב פולט)}
\textbf{התנגדות כניסה}
\[R_{in}=\left(\beta+1\right)\left(r_e+\left(r_o+R_C\right)\parallel R_L\right)\]
\textbf{התנגדות מוצא}
\[R_o=\frac{r_\pi+R_B}{\beta+1}\parallel\left(r_o+R_C\right)\]
\textbf{הגבר מתח}
\[\frac{v_o}{v_i}=\frac{r_o\parallel R_L}{r_o\parallel R_L+r_e}\]
\textbf{$r_{o}$$\to$$\infty$}
\[R_{in}=\left(\beta+1\right)\left(r_e+R_L'\right),\quad R_o=\frac{R_B}{\beta+1}+r_e,\quad \frac{i_o}{i_i}=\beta+1,\quad \frac{v_o}{v_i}=\frac{R_L'}{R_L'+r_e}\approx1\]
\subsection*{CB (בסיס משותף)}
\textbf{$r_{o}$$\to$$\infty$}
\[R_{in}=\frac{r_\pi}{\beta+1}=r_e=\frac{1}{g_m},\quad R_o\to\infty,\quad \frac{v_o}{v_i}=g_mR_L',\quad \frac{i_o}{i_i}=\alpha\]
\section{שיעור 6: ראי זרם ומקורות זרם}
\subsection*{ראי MOSFET}
\textbf{זרם ייחוס}
\[I_{REF}=\frac{V_{DD}-V_{GS}}{R}\]
\textbf{יחס הזרמים}
\[\frac{I_o}{I_{REF}}=\frac{\left(W/L\right)_2}{\left(W/L\right)_1}\]
\textbf{עם קיצור תעלה}
\[I_o=I_{REF}\frac{\left(W/L\right)_2}{\left(W/L\right)_1}\left(1+\lambda\left(V_o-V_{GS}\right)\right)\]
\textbf{התנגדות מוצא}
\[R_{out}=r_o=\frac{1}{\lambda I_D}=\frac{V_A}{I_o}\]
\textbf{הכוונת זרם}
\[I_2=I_{REF}\frac{\left(W/L\right)_2}{\left(W/L\right)_1},\qquad I_5=I_4\frac{\left(W/L\right)_5}{\left(W/L\right)_4}\]
\textbf{תנאי רוויה}
\[V_{D2},V_{D3}\ge -V_{SS}+V_{OV1},\qquad V_{D5}\le V_{DD}-|V_{OV5}|\]
\subsection*{ראי BJT}
\textbf{זרם עם נגד}
\[I_C=\frac{\beta}{\beta+2}\cdot\frac{V_{CC}-V_{BE}}{R}\approx\frac{V_{CC}}{R}\]
\textbf{יחס בסיסי}
\[\frac{I_o}{I_{REF}}=\frac{1}{1+2/\beta}=\frac{\beta}{\beta+2}\]
\textbf{הכפלה פי m}
\[\frac{I_o}{I_{REF}}=\frac{m}{1+\frac{m+1}{\beta}}\]
\textbf{כולל אפקט ארלי}
\[I_o=I_{REF}\frac{m}{1+\frac{m+1}{\beta}}\left(1+\frac{V_o-V_{BE}}{V_{A2}}\right)\]
\textbf{N יציאות}
\[I_C=I_{REF}\left(1+\frac{N+1}{\beta}\right)^{-1}\]
\textbf{זרם ייחוס בהכוונה}
\[I_{REF}=\frac{V_{CC}+V_{EE}-V_{EB1}-V_{BE2}}{R}\]
\subsection*{וילסון ווידלר}
\textbf{וילסון}
\[\frac{I_o}{I_{REF}}=\frac{1}{1+\frac{2}{\beta^2+2\beta}}\approx\frac{1}{1+2/\beta^2},\qquad R_{out}=\frac{\beta r_o}{2}\]
\textbf{וידלר}
\[I_oR_E=V_T\ln\frac{I_{REF}}{I_o}\]
\textbf{התנגדות מוצא וידלר}
\[R_{out}=r_{o2}\left(g_m\left(R_E\parallel r_\pi\right)+1\right)\]
\subsection*{קסקודה}
\textbf{MOS ללא עומס}
\[A_{v0}=-g_{m1}r_{o1}\left(1+g_{m2}r_{o2}\right)\approx-g_{m1}r_{o1}g_{m2}r_{o2}\]
\textbf{טרנזיסטורים תואמים}
\[A_{v0}=-\left(g_mr_o\right)^2\]
\textbf{התנגדות מוצא MOS}
\[R_o=\left(1+g_{m2}r_{o1}\right)r_{o2}+r_{o1}\approx g_{m2}r_{o2}r_{o1}\]
\textbf{MOS עם עומס}
\[G_v=-g_{m1}\left(g_{m2}r_{o2}r_{o1}\parallel R_L\right)=-G_m\left(R_o\parallel R_L\right)\]
\textbf{BJT}
\[R_{in}=r_\pi,\qquad R_o=\beta_2r_{o2},\qquad A_{v0}=-g_{m1}\beta_2r_{o2}\]
\textbf{BJT עם עומס (כפי שבסיכום)}
\[R_{out}=\left(\beta_2r_{o2}\right)\parallel R_L,\qquad G_v=-g_{m1}\left(r_{o2}\parallel R_L\right)\]
\section{שיעור 7: מגבר דיפרנציאלי}
\subsection*{הגדרות}
\textbf{כניסה הפרשית ומשותפת}
\[v_{id}=v^+-v^-,\qquad v_{CM}=\frac{v^++v^-}{2}\]
\textbf{מתח מוצא}
\[v_{out}=A_dv_{id}+A_{CM}v_{CM}\]
\textbf{CMRR}
\[\text{CMRR}=\left|\frac{A_d}{A_{CM}}\right|,\qquad \text{CMRR}_{dB}=20\log_{10}\left|\frac{A_d}{A_{CM}}\right|\]
\subsection*{זוג MOS}
\textbf{ענפים באיזון}
\[i_{D1}=i_{D2}=\frac{I}{2}\]
\textbf{תחום מתח משותף}
\[V_{SS}+V_{CS,\min}+V_{GS}\le V_{CM}\le V_{DD}-\frac12IR_D+V_t\]
\textbf{משוואות הזרמים}
\[\sqrt{i_{D1}}-\sqrt{i_{D2}}=v_{id}\sqrt{\tfrac12k_n'\tfrac{W}{L}},\qquad i_{D1}+i_{D2}=I\]
\textbf{תחום הכניסה}
\[\left|v_{id}\right|_{\max}=\sqrt2\,V_{OV}\]
\textbf{קירוב ליניארי}
\[i_{D1,2}=\frac I2\pm g_m\frac{v_{id}}{2},\qquad g_m=\frac{2I_D}{V_{OV}}=\frac{I}{V_{OV}}\]
\textbf{הגבר הפרשי}
\[A_d=\frac{v_{o2}-v_{o1}}{v_{id}}=g_m\left(r_o\parallel R_D\right)\]
\textbf{חד-צדדי}
\[A_{d,se}=\pm\frac12g_m\left(r_o\parallel R_D\right)\]
\textbf{הגבר משותף חד-צדדי}
\[A_{CM,se}=-\frac{R_D}{2R_{SS}},\qquad \text{CMRR}=g_mR_{SS}\]
\textbf{אי-התאמת $g_{m}$}
\[A_{CM,se}=\frac{R_D}{2R_{SS}}\frac{\Delta g_m}{g_m},\qquad \text{CMRR}=\frac{2g_m^2R_{SS}}{\Delta g_m}\]
\subsection*{זוג BJT}
\textbf{חלוקת זרם}
\[\frac{i_{E1,2}}{I}=0.5\pm\frac14\frac{v_{id}}{V_T}\]
